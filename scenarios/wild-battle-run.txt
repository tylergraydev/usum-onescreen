# Walk from the Pokémon Center to the tall grass south of it, wait for a
# wild battle, Run, and get back to the overworld.
include common/boot
# The panel path: with the native battle menu off, Battle gets the panel
# ([scenes] Battle = panel). native-wild-battle.txt covers the menu.
send battleui off
# Its X menu check expects the panel too: the native X menu (#22) off.
send xmenu off

# Out of the door: the map change logs scene (none), then FieldRo again.
mark
send stick 0 -150 120
expect-log ^scene: \(none\) 15
wait-scene FieldRo 30
sleep 6                   # the walk-out animation ignores the circle pad
# South down the path into the tall grass, in short steps: one long push
# stops partway (the walk halts and needs the pad back at neutral).
repeat 8 until-scene Battle
  send stick 0 -150 60
  sleep 1.5
end
shot grass
repeat 40 until-scene Battle  # back and forth in the grass
  send stick -150 0 20
  sleep 0.5
  send stick 150 0 20
  sleep 0.5
end
expect-mode panel         # Battle = panel
sleep 12                  # battle intro, until the command menu shows
shot battle
repeat 3 until-scene FieldRo
  send move 160 225; tap 6  # RUN (no A needed: it escapes at once)
  sleep 20                  # "Got away safely!" and the fade back to the field
end
expect-mode off           # FieldRo = off
shot field
# Outdoors the field state object sits elsewhere in the scene object than
# in the Pokémon Center: the X menu must still be seen, and its touch close.
sleep 3
send press x
expect-mode panel         # FieldRo.xmenu = panel
sleep 2
shot xmenu
send move 16 224; tap 6   # the X menu's back button
expect-mode off
stop
