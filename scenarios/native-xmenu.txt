# The native X menu (Screens/XMenu.cpp, docs/XMENU.md): X in the field shows
# it on the top screen with the panel off; every entry opens from it and
# comes back; B closes it. Each entry is picked with `xmenu select` (checked
# in the log) and a screenshot is taken before every A. Save stops at the
# game's confirm dialog and backs out with B; nothing is ever saved.
# Start in the Pokémon Center whatever ran before (native-move-learn leaves the grass).
restore pokecenter
include common/boot
# Checks the automatic view per scene, which is off by default.
send auto on

# --- Open: native menu, panel held off, the game's cursor read ---
mark
send press x
expect-log ^xmenu: shown 10
expect-log ^xmenu: game cursor page 10
expect-mode off           # FieldRo.xmenu would be panel; the native menu holds it off
sleep 1
shot menu

# --- Moving: D-pad and L/R move the native cursor, not the game's ---
mark
send xmenu select pokemon
expect-log ^xmenu: cursor pokemon 5
send press right
expect-log ^xmenu: cursor pokedex 5
send press right
expect-log ^xmenu: cursor pokemon_refresh 5
send press down
expect-log ^xmenu: cursor battle_video 5
send press down
expect-log ^xmenu: cursor save 5
send press left
expect-log ^xmenu: cursor map 5
send press up
expect-log ^xmenu: cursor festival_plaza 5
send press r
expect-log ^xmenu: cursor battle_video 5
send press l
expect-log ^xmenu: cursor bag 5
sleep 1
shot moved                # bottom: the game's own cursor hasn't moved
reject-log ^action:
expect-scene FieldRo

# --- Pokémon (Party) and back with B ---
mark
send xmenu select pokemon
expect-log ^xmenu: cursor pokemon 5

send press a
expect-log ^action: xmenu-pokemon ok 15
wait-scene main:00647FCC 10
# The native Party screen takes over (docs/PARTY-UI.md): panel off.
expect-log ^party-ui: shown 5
expect-mode off
sleep 2
shot party
mark
send press b
wait-scene FieldRo 20
expect-log ^xmenu: shown 15
expect-mode off
# Opening an entry moved the game's cursor there; the native one agrees.
expect-log ^xmenu: game cursor page 1 slot 0, native cursor pokemon 5

# needs a second tap. Tap only while Refresh is still up: the X menu's own
# close button sits at the same spot, and a tap that lands just after the
# field returns closes it (seen at 21:9).
