# Native battle menu (P2-1, W7-1, Screens/BattleScreen.hpp): wild battles
# played with buttons only, the panel off the whole time.
#   1. Fight -> move list -> B back; Pokémon -> the native list -> B back;
#      switch Pokémon (Pokémon, A, Switch) until the one coming in has been
#      hit; Bag -> HP/PP Restore -> Potion -> the hurt Pokémon (Potion x9 ->
#      x8); Run.
#   2. Bag -> Poké Balls -> Poké Ball, thrown until it is caught; B through
#      the Pokédex page / nickname question (No) to the field.
#   3. Fight, a different move each turn, until the battle is won.
# Every choice is a press; the menu taps the hidden bottom screen itself.
# Never saves.
restore pokecenter
include common/boot

send battleui
expect-log ^battle-ui: on 5

# Out of the door and into the tall grass (as wild-battle-run.txt).
mark
send stick 0 -150 120
expect-log ^scene: \(none\) 15
wait-scene FieldRo 30
sleep 6
repeat 8 until-scene Battle
  send stick 0 -150 60
  sleep 1.5
end
mark
repeat 40 until-scene Battle
  send stick -150 0 20
  sleep 0.5
  send stick 150 0 20
  sleep 0.5
end

# --- Battle 1 ---
expect-mode off           # native battle menu: no panel in battle
expect-log ^battle-ui: page command, cursor 0 of 4 \(Fight\) 40
shot command
send press a              # Fight
expect-log ^action: battle-fight ok 10
expect-log ^battle-ui: page moves, cursor \d of 4 \((Scratch|Fire Fang|Double Kick|Lick), 5
expect-log ^  player move names: Scratch, Fire Fang, Double Kick, Lick 10
sleep 1
shot moves
send press b              # back to the commands
expect-log ^action: battle-moves-back ok 10
sleep 1
send press down
sleep 1
send press down           # Pokémon
sleep 1
shot command-pokemon
send press a
expect-log ^action: battle-pokemon ok 10
expect-log ^battle-ui: page party, cursor 1 of 6 \(Zorua Lv14 10
sleep 1
shot pokemon-list
mark
send press b              # back: the list's back button
expect-log ^battle-ui: step battle_party\.back 10
expect-log ^battle-ui: page command, cursor 0 of 4 \(Fight\)$ 15

# Switch until the Pokémon that comes in has been hit (the foe attacks it
# as it comes in): Zorua and Torracat take turns at the front.
repeat 6 until-log ^\s+player\s\w+\ssp\s(570\sL\d+\shp\s([0-9]|[12]\d|3[0-4])/35|726\sL\d+\shp\s([0-9]|[1-4]\d|50)/51)\s
  expect-log ^battle-ui: page command, cursor 0 of 4 \(Fight\)$ 40
  send press down
  sleep 1
  send press down         # Pokémon
  sleep 1
  send press a
  expect-log ^battle-ui: page party, cursor 1 of 6 \((Zorua|Torracat) 15
  sleep 1
  send press a
  expect-log ^battle-ui: page party, cursor 1 of 6 .*, menu Switch$ 5
  shot switch-menu
  send press a            # Switch
  expect-log ^battle-ui: step party\.slot2 15
  expect-log ^battle-ui: step battle_party_page\.switch 15
  mark
  sleep 10
end

# A Potion on it: the Bag (first tab in a new battle), Right to HP/PP Restore,
# A on Potion; the pick starts on the hurt Pokémon.
send bag
expect-log ^  Medicine \d+/60: Potion\(17\) x9 5
expect-log ^battle-ui: page command, cursor 0 of 4 \(Fight\)$ 40
mark
send press down           # Bag
sleep 1
send press a
expect-log ^action: battle-bag ok 10
expect-log ^battle-ui: page bag, Rotom Powers, cursor 0 of 1 10
sleep 1
send press right
expect-log ^battle-ui: step battle_bag\.pocket_hp_pp 10
expect-log ^battle-ui: page bag, HP/PP Restore, cursor 0 of 4 \(Potion x9\)$ 10
sleep 1
shot bag-hp
send press a
expect-log ^battle-ui: step bag\.item_row1 10
expect-log ^battle-ui: step battle_bag_item_page\.use 10
expect-log ^battle-ui: page bag, HP/PP Restore, pick \d of 6 \((Zorua|Torracat) (\d+)/(35|51)\)$ 10
sleep 1
shot bag-pick
send press a
expect-log ^battle-ui: page bag, HP/PP Restore, pick .*, busy: Using Potion on 5
expect-log ^battle-ui: step bag_party\.slot 10
expect-log ^battle-ui: page command, cursor 0 of 4 \(Fight\)$ 40
send bag
expect-log ^  Medicine \d+/60: Potion\(17\) x8 5
reject-log ^battle-ui: .*note:
sleep 1
send press up             # wraps to Run
sleep 1
shot command-run
send press a
expect-log ^battle-ui: chose battle-flee 10
# Run is refused while the foe has trapped us (seen: Ekans's Wrap); then
# the battle is fought out with moves (any level-up move question is
# skipped).
repeat 12 until-scene FieldRo
  sleep 8
  if-scene Battle send action battle-fight
  sleep 1.5
  if-scene Battle send action battle-move1
  sleep 6
  if-scene Battle send press b
  sleep 2
  if-scene Battle send press b
  if-scene Battle send action learn-skip
end
wait-scene FieldRo 30
expect-mode off
sleep 3

# --- Battle 2: catch ---
mark
repeat 40 until-scene Battle
  send stick -150 0 20
  sleep 0.5
  send stick 150 0 20
  sleep 0.5
end
expect-mode off
expect-log ^battle-ui: page command, cursor 0 of 4 \(Fight\) 40
send press down           # Bag
sleep 1
send press a
expect-log ^action: battle-bag ok 10
# A new battle's Bag opens on its first tab; Poké Balls is the fourth
# (within a battle it keeps its tab).
expect-log ^battle-ui: page bag, Rotom Powers, cursor 0 of 1 10
sleep 1
send press right
expect-log ^battle-ui: step battle_bag\.pocket_hp_pp 10
sleep 1.5
send press right
expect-log ^battle-ui: step battle_bag\.pocket_status 10
sleep 1.5
send press right
expect-log ^battle-ui: page bag, Poké Balls, cursor 0 of 3 \(Poké Ball x\d+\)$ 10
sleep 1
shot balls
send press a              # throw it
expect-log ^battle-ui: page bag, Poké Balls, .*busy: Throwing Poké Ball 5
expect-log ^battle-ui: step battle_bag_item_page\.use 15
sleep 30                  # the throw, the shakes, the opponent's turn
# Broke free: the same again (the Bag keeps its tab).
repeat 12 until-scene Zukan
  expect-log ^battle-ui: page command, cursor 0 of 4 \(Fight\)$ 20
  mark
  send press down
  sleep 1
  send press a
  expect-log ^battle-ui: page bag, Poké Balls, cursor 0 of 3 \(Poké Ball x\d+\)$ 10
  sleep 1
  shot balls-again
  send press a
  expect-log ^battle-ui: step battle_bag_item_page\.use 15
  sleep 30
end
wait-scene Zukan 30
expect-mode off           # the Pokédex scene straight after the battle
expect-log ^battle-ui: page game, after catch 10
sleep 2
shot caught
# A new species: its Pokédex page, then (any species) the nickname
# question, where B is No, then "sent to a Box". B through all of it.
repeat 10 until-scene FieldRo
  shot after-catch
  send press b
  sleep 3
end
wait-scene FieldRo 30
expect-mode off
sleep 3

# --- Battle 3: win with moves ---
mark
repeat 40 until-scene Battle
  send stick -150 0 20
  sleep 0.5
  send stick 150 0 20
  sleep 0.5
end
expect-mode off
repeat 8 until-scene FieldRo
  # Since the last turn's mark: the bar as shown, not yet "sent".
  expect-log ^battle-ui: page command, cursor 0 of 4 \(Fight\)$ 40
  send press a            # Fight
  expect-log ^battle-ui: page moves, cursor \d of 4 10
  sleep 1
  # The list starts on the last move used; take the next one each turn, so
  # a Disable (Drowzee has it: it blocks the last move used) never stops us.
  send press down
  sleep 1
  send press a
  expect-log ^action: battle-move\d ok 10
  mark
  # The turn, then EXP and level-up windows (Exp. Share): B advances them.
  # The catch's EXP can make Buneary want a fifth move (docs/MOVE-LEARN.md):
  # learn-skip keeps the old moves; it is refused when the question isn't up.
  repeat 7
    sleep 4
    if-scene Battle send press b
  end
  if-scene Battle send action learn-skip
  sleep 3
  repeat 3
    sleep 3
    if-scene Battle send press b
  end
end
expect-mode off
shot field
reject-log ^action: .* FAILED
stop
