# Every native X menu entry on the Ultra Moon postgame save, whose X menu is
# not the default layout (docs/XMENU.md "Layout"): page 1 Pokémon, Pokédex,
# Bag, Photo Album, Quick Link, Festival Plaza; page 2 Trainer Passport,
# Poké Pelago, Pokémon Refresh, QR Scanner, Battle Video, Options. Each entry
# is picked by name, opened with A, and must open its own scene (or, for the
# entries without one, close the X menu), then is backed out of. Runs on an
# Ultra Moon instance only (tools/instance.py game NAME moon). Nothing is
# saved. Poké Pelago comes last: its first visit starts a tutorial that wants
# taps, so the run stops there.
restore umoon-postgame
include common/boot

# --- The native menu reads the game's layout ---
mark
send press x
expect-log ^xmenu: layout p1 pokemon pokedex bag photo_album quick_link festival_plaza \| p2 trainer_passport poke_pelago pokemon_refresh qr_scanner battle_video options 10
expect-log ^xmenu: shown 10
expect-mode off
sleep 1
shot menu

# --- Pokémon ---
mark
send xmenu select pokemon
expect-log ^xmenu: cursor pokemon 5
send press a
expect-log ^action: xmenu-pokemon ok 15
wait-scene main:00647FCC 10
expect-log ^party-ui: shown 5
sleep 2
shot pokemon
mark
send press b
wait-scene FieldRo 20
expect-log ^xmenu: shown 15

# --- Pokédex (its back button returns to the X menu) ---

wait-scene Zukan 10
sleep 3

mark
send move 15 223; tap 6
wait-scene FieldRo 20
expect-log ^xmenu: shown 15
